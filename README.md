<script>
    // 1. Aquí colocas la lista con los nombres exactos de tus archivos de imagen
    const misImagenes = [
        "imagenes/foto1.jpg",
        "imagenes/foto2.jpg",
        "imagenes/foto3.jpg",
        "imagenes/foto4.jpg",
        "imagenes/foto5.jpg"
    ];

    // Guardamos la última posición elegida para evitar que salga la misma dos veces seguidas
    let ultimoIndice = 0;

    // Seleccionamos los elementos del HTML
    const imgElement = document.getElementById('imagenAleatoria');
    const btnElement = document.getElementById('btnCambiar');

    // Función principal para mezclar y seleccionar
    function mezclarYSeleccionar() {
        let nuevoIndice;

        // Bucle para asegurarnos de que la nueva imagen no sea igual a la anterior
        do {
            nuevoIndice = Math.floor(Math.random() * misImagenes.length);
        } while (nuevoIndice === ultimoIndice && misImagenes.length > 1);

        // Guardamos el índice actual para la siguiente vuelta
        ultimoIndice = nuevoIndice;

        // Cambiamos la imagen en el HTML
        imgElement.src = misImagenes[nuevoIndice];
    }

    // Escuchamos el clic del botón
    btnElement.addEventListener('click', mezclarYSeleccionar);
</script>
